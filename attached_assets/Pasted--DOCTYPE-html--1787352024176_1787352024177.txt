<script>
  /* ========================================================================
     CONFIGURAÇÃO — cole aqui o endpoint do seu serviço de formulário.
     Vazio = modo demonstração (mostra sucesso, mas NÃO envia).
     Ex.: "https://formspree.io/f/xxxxxxx"
     ======================================================================== */
  const WAITLIST_ENDPOINT = "";

  const form = document.getElementById('waitlist');
  const formView = document.getElementById('form-view');
  const successView = document.getElementById('success-view');
  const successMsg = document.getElementById('success-msg');
  const submitBtn = document.getElementById('submit-btn');

  function showSuccess(intencao){
    formView.style.display = 'none';
    successView.style.display = 'block';
    if (intencao === 'ajudar' || intencao === 'os-dois') {
      successMsg.textContent = 'A gente te avisa quando abrir. E que bom que você quer ajudar — é gente como você que faz esse lugar existir. 💛';
    }
    successView.scrollIntoView({behavior:'smooth', block:'center'});
  }

  form.addEventListener('submit', async function(e){
    e.preventDefault();
    const data = new FormData(form);
    const intencao = data.get('intencao');
    submitBtn.disabled = true;
    submitBtn.textContent = 'Enviando...';

    // Modo demonstração: sem endpoint configurado, só mostra o sucesso.
    if (!WAITLIST_ENDPOINT) {
      setTimeout(() => showSuccess(intencao), 500);
      return;
    }

    try {
      const res = await fetch(WAITLIST_ENDPOINT, {
        method: 'POST',
        headers: { 'Accept': 'application/json' },
        body: data
      });
      if (res.ok) {
        showSuccess(intencao);
      } else {
        throw new Error('Falha no envio');
      }
    } catch (err) {
      submitBtn.disabled = false;
      submitBtn.textContent = 'Entrar na fila';
      alert('Não conseguimos enviar agora. Tente de novo em instantes.');
    }
  });
</script>
